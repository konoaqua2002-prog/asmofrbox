# syntax=docker/dockerfile:1
FROM mcr.microsoft.com/dotnet/sdk:8.0 AS build
WORKDIR /src
COPY . .
RUN dotnet publish AsmoFrBoxTelegramBot.csproj -c Release -o /app

FROM mcr.microsoft.com/dotnet/runtime:8.0 AS runtime
WORKDIR /app
COPY --from=build /app .
# TELEGRAM_BOT_TOKEN is required at runtime; CATALOG_SOURCE is optional.
ENTRYPOINT ["dotnet", "AsmoFrBoxTelegramBot.dll"]
